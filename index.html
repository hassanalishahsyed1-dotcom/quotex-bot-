import tkinter as tk from tkinter import ttk import yfinance as yf import pandas as pd import threading import time class QuotexRealEngineBot: def __init__(self, root): self.root = root self.root.title("Quotex Verified Signal Bot") self.root.geometry("400x520") self.root.configure(bg="#0f172a") # ڈارک نائٹ تھیم self.is_running = False # --- UI انٹرفیس --- tk.Label(root, text="QUOTEX REAL ENGINE", font=("Arial", 18, "bold"), fg="#38bdf8", bg="#0f172a").pack(pady=20) # اثاثہ (Asset) tk.Label(root, text="Select Real Market Asset:", font=("Arial", 11), fg="#94a3b8", bg="#0f172a").pack() self.asset_box = ttk.Combobox(root, values=["EURUSD", "GBPUSD", "USDJPY", "AUDUSD"], font=("Arial", 12), state="readonly") self.asset_box.set("EURUSD") self.asset_box.pack(pady=5) # بٹن self.start_btn = tk.Button(root, text="ACTIVATE BOT", font=("Arial", 12, "bold"), bg="#38bdf8", fg="#0f172a", bd=0, width=18, height=1, command=self.toggle_bot) self.start_btn.pack(pady=20) # سگنل پینل self.signal_frame = tk.Frame(root, bg="#1e293b", bd=2, relief="solid") self.signal_frame.pack(pady=10, fill="x", padx=40) self.signal_text = tk.Label(self.signal_frame, text="READY", font=("Arial", 26, "bold"), fg="#64748b", bg="#1e293b", pady=25) self.signal_text.pack() # لائیو کینڈل انفارمیشن self.candle_info = tk.Label(root, text="Last Candle: No Data", font=("Arial", 11), fg="#94a3b8", bg="#0f172a") self.candle_info.pack(pady=10) self.status_label = tk.Label(root, text="System: Idle", font=("Arial", 9), fg="#64748b", bg="#0f172a") self.status_label.pack(side="bottom", pady=10) def toggle_bot(self): if not self.is_running: self.is_running = True self.start_btn.config(text="DEACTIVATE", bg="#ef4444", fg="white") self.status_label.config(text="System: Connecting to real market feeds...") threading.Thread(target=self.core_engine_loop, daemon=True).start() else: self.is_running = False self.start_btn.config(text="ACTIVATE BOT", bg="#38bdf8", fg="#0f172a") self.status_label.config(text="System: Idle") self.signal_text.config(text="READY", fg="#64748b") self.candle_info.config(text="Last Candle: No Data") def core_engine_loop(self): """یہ اس بوٹ کا اصل دماغ ہے جو فرضی سگنل نہیں بناتا""" while self.is_running: asset = self.asset_box.get() try: # انٹرنیٹ سے لائیو 1 منٹ کی کینڈلز کا ڈیٹا لانا ticker = yf.Ticker(f"{asset}=X") df = ticker.history(period="1d", interval="1m") if not df.empty and len(df) >= 3: df.columns = [col.lower() for col in df.columns] # آخری بند ہونے والی مکمل کینڈل کا ڈیٹا نکالیں (تاکہ سگنل ریپینٹ نہ ہو) last_candle = df.iloc[-2] open_p = last_candle['open'] close_p = last_candle['close'] high_p = last_candle['high'] low_p = last_candle['low'] # 🟢 سچی کینڈل اسٹک لاجک (Price Action & Volume Spread Analysis) # کینڈل کی باڈی اور وِک (Wick) کا سائز معلوم کرنا body_size = abs(close_p - open_p) total_range = high_p - low_p self.candle_info.config(text=f"Last Close: {close_p:.5f}\nBody: {body_size:.5f}") # 1. ہیمر / پن بار پیٹرن (بائے سگنل) - اگر نیچے سے ریجیکشن ہو if close_p > open_p and (open_p - low_p) > (body_size * 2): self.signal_text.config(text="🟢 UP (CALL)", fg="#22c55e") self.status_label.config(text="Reason: Strong Buyer Rejection from Low") # 2. شوٹنگ اسٹار پیٹرن (سیل سگنل) - اگر اوپر سے ریجیکشن ہو elif close_p < open_p and (high_p - open_p) > (body_size * 2): self.signal_text.config(text="🔴 DOWN (PUT)", fg="#ef4444") self.status_label.config(text="Reason: Strong Seller Rejection from High") # 3. ماروبوزو کینڈل (مضبوط مومینٹم) elif close_p > open_p and body_size > (total_range * 0.8): self.signal_text.config(text="🟢 UP (CALL)", fg="#22c55e") self.status_label.config(text="Reason: Strong Bullish Momentum") elif close_p < open_p and body_size > (total_range * 0.8): self.signal_text.config(text="🔴 DOWN (PUT)", fg="#ef4444") self.status_label.config(text="Reason: Strong Bearish Momentum") else: # اگر مارکیٹ سائیڈ ویز یا کنفیوزڈ ہے، تو کوئی فرضی سگنل نہیں دے گا self.signal_text.config(text="NO SIGNAL", fg="#eab308") self.status_label.config(text="Reason: Market is unstable, waiting for clear pattern.") except Exception as e: print(f"Error: {e}") self.status_label.config(text="System: Connection unstable. Retrying...") # کوٹیکس پر 1 منٹ کی کینڈل ہوتی ہے، اس لیے یہ ہر 10 سیکنڈ بعد لائیو مارکیٹ چیک کرے گا time.sleep(10) if __name__ == "__main__": root = tk.Tk() bot = QuotexRealEngineBot(root) root.mainloop()
